```rust
/// Convert SpliceError to ErrorCode for JSON output.
fn error_code_from_splice_error(error: &splice::SpliceError) -> splice::error_codes::ErrorCode {
    use splice::error_codes::{ErrorCode, SpliceErrorCode};
    use splice::SpliceError;

    // Extract error context
    let file = error.file_path().and_then(|p| p.to_str()).map(|s| s.to_string());
    let line = error.line();
    let column = error.column();

    // Map SpliceError to SpliceErrorCode
    let splice_code = match error {
        SpliceError::SymbolNotFound { .. } => SpliceErrorCode::SymbolNotFound,
        SpliceError::InvalidSpan { .. } => SpliceErrorCode::InvalidSpan,
        SpliceError::ParseError { .. } => SpliceErrorCode::ParseError,
        SpliceError::IoError(_) => SpliceErrorCode::IoError,
        SpliceError::GraphError(_) => SpliceErrorCode::GraphError,
        SpliceError::ValidationError { .. } => SpliceErrorCode::ValidationError,
        SpliceError::BackupError(_) => SpliceErrorCode::BackupError,
        SpliceError::ExecutionNotFound { .. } => SpliceErrorCode::ExecutionNotFound,
        SpliceError::Other(_) => SpliceErrorCode::OtherError,
        _ => SpliceErrorCode::OtherError,
    };

    ErrorCode::from_splice_code(splice_code, file.as_deref(), line, column)
}
```

Why: Provides centralized conversion from SpliceError to ErrorCode with proper location extraction.
