    Pattern for both commands:
    ```rust
    let mut cache = RelationshipCache::new();
    if cli.relationships {
        let callers = get_callers(&graph, resolved.node_id, &mut cache);

        let exports = get_exports(&graph, &file_path, &mut cache);

        span_result = span_result.with_relationships(relationships);
    ```
