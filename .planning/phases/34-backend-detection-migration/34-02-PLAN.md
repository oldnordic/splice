    **Part 1: Update src/cli/mod.rs**

    1. In the Status command (around line 474), add the detect_backend flag:
       ```rust
    /// Show database statistics (files, symbols, refs, calls, chunks)
    #[command(display_order = 100)]
    Status {
        /// Path to the Magellan database
        #[arg(short, long)]
        db: std::path::PathBuf,

        /// Detect and report the backend format (sqlite or native-v2)
        #[arg(long, default_value = "false")]
        detect_backend: bool,
    },
       ```

    **Part 2: Update src/main.rs**

    2. Find the execute_status function pattern match in main() - look for:
       ```rust
    splice::cli::Commands::Status { db } => execute_status(&db, json_output),
       ```
       Change to:
       ```rust
    splice::cli::Commands::Status { db, detect_backend } => {
        execute_status(&db, json_output, detect_backend)
    }
       ```

    3. Find the execute_status function definition and update its signature:
       - Change from: `fn execute_status(db_path: &Path, json_output: bool) -> Result<...>`
       - Change to: `fn execute_status(db_path: &Path, json_output: bool, detect_backend: bool) -> Result<...>`

    4. Add the following logic at the START of execute_status (before any database operations):
       ```rust
    // If detect_backend flag is set, report backend and exit early
    if detect_backend {
        let backend = splice::graph::CodeGraph::detect_backend(db_path)?;
        if json_output {
            let output = json!({
                "backend": backend.to_string(),
                "database": db_path.to_string_lossy(),
            });
            println!("{}", output);
        } else {
            println!("Backend: {}", backend);
            println!("Database: {}", db_path.display());
        }
        return Ok(splice::cli::CliSuccessPayload::default());
    }

    // Check for native-v2 database without native-v2 feature
    let detected_backend = splice::graph::CodeGraph::detect_backend(db_path)?;
    if detected_backend == splice::graph::Backend::NativeV2 {
        #[cfg(not(feature = "native-v2"))]
        {
            return Err(splice::SpliceError::Other(
                "Cannot open native-v2 database without native-v2 feature. \
                 Build with: cargo build --features native-v2 --no-default-features".to_string()
            ));
        }
    }
       ```

    5. Add necessary imports at top of main.rs if not present:
       ```rust
    use splice::graph::Backend;
       ```
