#[command(name = "splice")]
#[command(author, version, about, long_about = None)]

Update error handling in main():
```rust
if let Err(e) = run(cli) {
    if cli.json_output() {
        let error_result = OperationResult {
            version: SCHEMA_VERSION.to_string(),
            operation_id: Uuid::new_v4().to_string(),
            operation_type: operation_type_from_command(&cli.command),
            status: "error".to_string(),
            message: e.to_string(),
            timestamp: chrono::Utc::now().to_rfc3339(),

            result: None,
            error: Some(ErrorDetails::from_splice_error(&e)),

        println!("{}", serde_json::to_string_pretty(&error_result).unwrap());

        eprintln!("Error: {}", e);
    }
    std::process::exit(1);
}
