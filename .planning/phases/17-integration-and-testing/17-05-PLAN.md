//! Cross-tool alignment tests with Magellan format compatibility.
//!
//! Tests verify that Splice's MagellanIntegration wrapper correctly uses
//! the Magellan v0.5.3 API for cross-tool compatibility.
//!
//! API surface tested:
//! - index_file() - Ingest files into Magellan database
//! - query_by_labels() - Retrieve code chunks by label patterns
//! - get_code_chunk() - Retrieve code by byte range
//!
//! Note: These tests validate Splice's correct usage of Magellan API.
//! External database compatibility is Magellan's responsibility.

use splice::graph::magellan_integration::MagellanIntegration;
use splice::symbol::Language;
use std::io::Write;
use tempfile::TempDir;

/// Helper to create a test Magellan integration instance.
fn create_test_magellan_integration() -> (MagellanIntegration, TempDir) {
    let temp_dir = TempDir::new().expect("Failed to create temp dir");
    let db_path = temp_dir.path().join("test_magellan.db");

    let integration =
        MagellanIntegration::new(&db_path).expect("Failed to create MagellanIntegration");

    (integration, temp_dir)
}

/// Helper to create a test source file.
fn create_test_file(dir: &Path, name: &str, content: &str) -> std::path::PathBuf {
    let file_path = dir.join(name);
    let mut file = std::fs::File::create(&file_path).unwrap();
    file.write_all(content.as_bytes()).unwrap();
    file_path
}

/// Test MagellanIntegration::new() creates database.
#[test]
fn test_magellan_integration_new() {
    let temp_dir = TempDir::new().expect("Failed to create temp dir");
    let db_path = temp_dir.path().join("test_magellan.db");

    // Create integration
    let integration = MagellanIntegration::new(&db_path);

    assert!(integration.is_ok(), "MagellanIntegration::new should succeed");

    let integration = integration.unwrap();

    // Database file should exist
    assert!(db_path.exists(), "Database file should be created");

    println!("MagellanIntegration created successfully at: {}", db_path.display());
}

/// Test index_file() ingests file into Magellan database.
#[test]
fn test_magellan_index_file() {
    let (integration, temp_dir) = create_test_magellan_integration();

    // Create test file
    let content = r#"
/// Adds two numbers
pub fn add(a: i32, b: i32) -> i32 {
    a + b
}

/// Multiplies two numbers
pub fn multiply(a: i32, b: i32) -> i32 {
    a * b
}
"#;

    let file_path = create_test_file(temp_dir.path(), "test.rs", content);

    // Index the file
    let result = integration.index_file(&file_path, Language::Rust);

    assert!(result.is_ok(), "index_file should succeed: {:?}", result.err());

    let indexed_count = result.unwrap();
    assert!(indexed_count > 0, "Should index at least one symbol");

    println!("Indexed {} symbols from test.rs", indexed_count);
}

/// Test query_by_labels() retrieves code chunks by labels.
#[test]
fn test_magellan_query_by_labels() {
    let (integration, temp_dir) = create_test_magellan_integration();

    // Create and index test file
    let content = r#"
pub fn function_one() -> i32 {
    42
}

pub fn function_two() -> String {
    "hello".to_string()
}
"#;

    let file_path = create_test_file(temp_dir.path(), "test.rs", content);
    integration.index_file(&file_path, Language::Rust).unwrap();

    // Query by label pattern
    let labels = vec!["function_one".to_string(), "function_two".to_string()];
    let result = integration.query_by_labels(&labels);

    assert!(result.is_ok(), "query_by_labels should succeed: {:?}", result.err());

    let chunks = result.unwrap();
    assert!(chunks.len() > 0, "Should retrieve at least one code chunk");

    println!("Retrieved {} code chunks by labels", chunks.len());

    // Verify chunk structure
    for chunk in &chunks {
        assert!(!chunk.content.is_empty(), "Chunk content should not be empty");
        assert!(chunk.byte_start < chunk.byte_end, "Byte range should be valid");
    }
}

/// Test get_code_chunk() retrieves code by byte range.
#[test]
fn test_magellan_get_code_chunk() {
    let (integration, temp_dir) = create_test_magellan_integration();

    // Create and index test file
    let content = r#"
pub fn example_function() -> i32 {
    100
}
"#;

    let file_path = create_test_file(temp_dir.path(), "test.rs", content);
    integration.index_file(&file_path, Language::Rust).unwrap();

    // Get code chunk by byte range
    let byte_start = 10;
    let byte_end = 50;

    let result = integration.get_code_chunk(&file_path, byte_start, byte_end);

    assert!(result.is_ok(), "get_code_chunk should succeed: {:?}", result.err());

    let chunk = result.unwrap();
    assert!(!chunk.content.is_empty(), "Chunk content should not be empty");
    assert_eq!(chunk.byte_start, byte_start);
    assert_eq!(chunk.byte_end, byte_end);

    println!("Retrieved code chunk: {} bytes", chunk.content.len());
}

/// Test query_by_labels() with non-existent labels returns empty result.

    // Query for labels that don't exist

    let result = integration.query_by_labels(&labels);

    assert!(result.is_ok(), "query_by_labels should succeed even with no matches");

    let chunks = result.unwrap();
    assert_eq!(chunks.len(), 0, "Should return empty array for non-existent labels");

    println!("Correctly returned empty result for non-existent labels");
}

        create_test_file(temp_dir.path(), "test.ts", typescript_content);

    // Index all files
    let rust_count = integration.index_file(&rust_file, Language::Rust).unwrap();
    let python_count = integration.index_file(&python_file, Language::Python).unwrap();

    assert!(rust_count > 0, "Should index Rust symbols");
    assert!(python_count > 0, "Should index Python symbols");

    println!(
        "Indexed {} Rust, {} Python, {} TypeScript symbols",
        rust_count, python_count, typescript_count
    );
}

/// Test cross-tool format compatibility: chunk structure matches expected format.
#[test]
fn test_magellan_chunk_format_compatibility() {
    let (integration, temp_dir) = create_test_magellan_integration();

    // Create and index test file
    let content = "pub fn compatibility_test() -> i32 { 123 }";
    let file_path = create_test_file(temp_dir.path(), "test.rs", content);
    integration.index_file(&file_path, Language::Rust).unwrap();

    // Get code chunk
    let result = integration.get_code_chunk(&file_path, 0, content.len());

    assert!(result.is_ok(), "get_code_chunk should succeed");

    let chunk = result.unwrap();

    // Verify chunk has required fields for cross-tool compatibility
    assert!(!chunk.file_path.is_empty(), "Chunk should have file_path");
    assert!(!chunk.content.is_empty(), "Chunk should have content");
    assert!(chunk.byte_start < chunk.byte_end, "Chunk should have valid byte range");

    // Verify file_path is absolute (required for cross-tool compatibility)
    assert!(
        chunk.file_path.is_absolute(),
        "file_path should be absolute for cross-tool compatibility"
    );

    println!(
        "Chunk format compatible: file={}, bytes={}..{}",
        chunk.file_path.display(),
        chunk.byte_start,
        chunk.byte_end
    );
}
