//! Performance tests for relationship queries on large codebases.

//! - get_callers: < 100ms

//! meaningful performance validation for relationship query infrastructure.

use splice::graph::CodeGraph;
use splice::relationships::{Relationship, RelationshipCache, get_callers, get_callees, get_imports, get_exports};

/// Helper to create a test code graph with specified number of symbols.
/// Reuses TestGraphBuilder pattern from existing relationship_performance.rs.
struct TestGraphBuilder {
    graph: CodeGraph,
    temp_dir: TempDir,
    symbols_per_file: usize,
}

impl TestGraphBuilder {
    /// Create a new test graph builder.
    fn new() -> std::io::Result<Self> {
        let temp_dir = TempDir::new()?;
        let db_path = temp_dir.path().join("test_rel_perf.db");
        let graph = CodeGraph::open(&db_path).expect("Failed to open graph");

        Ok(Self {
            graph,
            temp_dir,
            symbols_per_file: 10,
        })
    }

    /// Get mutable reference to the graph.
    fn graph_mut(&mut self) -> &mut CodeGraph {
        &mut self.graph
    }

    /// Create a file with symbols and ingest into graph.
    fn create_file_with_symbols(
        &mut self,
        file_index: usize,
        num_symbols: usize,
    ) -> std::io::Result<std::path::PathBuf> {
        let file_path = self
            .temp_dir
            .path()
            .join(format!("test_file_{}.rs", file_index));

        let mut file = std::fs::File::create(&file_path)?;

        // Write symbol definitions
        for i in 0..num_symbols {
            let symbol_name = format!("function_{}_{}", file_index, i);
            let source_code = format!(
                r#"
/// Documentation for {}
pub fn {}() {{
    // Implementation
}}
"#,
                symbol_name, symbol_name
            );

            file.write_all(source_code.as_bytes())?;

            // Store symbol in graph
            let byte_start = source_code.len() * i;
            let byte_end = byte_start + source_code.len();
            let line_start = i * 5 + 1;
            let line_end = line_start + 4;

            self.graph
                .store_symbol_with_file_and_language(
                    &file_path,
                    &symbol_name,
                    "function",
                    byte_start,
                    byte_end,
                    line_start,
                    line_end,
                    Language::Rust,
                )
                .expect("Failed to store symbol");
        }

        Ok(file_path)
    }

    /// Create a large graph with specified total symbols.
    fn create_large_graph(&mut self, total_symbols: usize) -> std::io::Result<()> {
        let num_files = total_symbols / self.symbols_per_file;

        for file_index in 0..num_files {
            self.create_file_with_symbols(file_index, self.symbols_per_file)?;
        }

        Ok(())
    }

    /// Get the graph and temp dir (consumes builder).
    fn build(self) -> (CodeGraph, TempDir) {
        (self.graph, self.temp_dir)
    }
}

/// Test get_callers performance on 1K symbol graph.
#[test]
fn test_get_callers_1k_graph_performance() {
    let (graph, temp_dir) = create_large_graph(1000); // 1K symbols

    let file_path = temp_dir.path().join("test_file_0.rs");
    let file_path_str = file_path.to_str().expect("Invalid path");

    // Find a symbol
    let node_id = graph
        .find_symbol_in_file(file_path_str, "function_0_0")
        .expect("Symbol not found");

    let mut cache = RelationshipCache::new();
    let start = Instant::now();

    let result = get_callers(&graph, node_id, &mut cache);

    let duration = start.elapsed();

    assert!(result.is_ok(), "get_callers failed: {:?}", result.err());

    // Verify performance: < 100ms
    assert!(
        duration.as_millis() < 100,
        "get_callers on 1K graph took {}ms, expected < 100ms",
        duration.as_millis()
    );

    println!("get_callers on 1K graph: {}ms", duration.as_millis());
}

        .expect("Symbol not found");

    // First query (no cache)
    let mut cache = RelationshipCache::new();
    let start = Instant::now();
    let _ = get_callers(&graph, node_id, &mut cache).unwrap();
    let first_duration = start.elapsed();

    // Second query (cached)
    let start = Instant::now();
    let _ = get_callers(&graph, node_id, &mut cache).unwrap();
    let cached_duration = start.elapsed();

    // Cached query should be faster (or at least not slower)
    println!(
        "First query: {}ms, Cached query: {}ms",
        first_duration.as_millis(),
        cached_duration.as_millis()

    // Cached query should be significantly faster or at least not slower
    // We allow some tolerance for system load
    assert!(
        cached_duration.as_millis() <= first_duration.as_millis() + 10,
        "Cached query should not be significantly slower"
    );
}
