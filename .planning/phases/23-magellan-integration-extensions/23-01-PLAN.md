```rust
/// Get comprehensive database statistics.
///
/// Returns counts of all entity types in the graph database.
pub fn get_statistics(&self) -> Result<DatabaseStats> {
    let files = self.inner.count_files()
        .map_err(|e| SpliceError::Other(format!("Failed to count files: {}", e)))?;
    let symbols = self.inner.count_symbols()
        .map_err(|e| SpliceError::Other(format!("Failed to count symbols: {}", e)))?;
    let references = self.inner.count_references()
        .map_err(|e| SpliceError::Other(format!("Failed to count references: {}", e)))?;
    let code_chunks = self.inner.count_chunks()
        .map_err(|e| SpliceError::Other(format!("Failed to count code chunks: {}", e)))?;

    // Magellan has no count_calls() method - count Call nodes explicitly
    let calls = self.count_call_nodes()?;

    Ok(DatabaseStats {
        files,
        symbols,
        references,
        calls,
        code_chunks,
    })
}
```

3. Add private helper method count_call_nodes():
```rust
/// Count Call nodes by iterating all entities (Magellan lacks count_calls() API).
fn count_call_nodes(&self) -> Result<usize> {
    let entity_ids = self.inner.entity_ids()
        .map_err(|e| SpliceError::Other(format!("Failed to get entity IDs: {}", e)))?;

    let call_count = entity_ids.iter()
        .filter_map(|id| self.inner.get_node(*id).ok())
        .filter(|node| node.kind == "Call")
        .count();

    Ok(call_count)
}
```
