Find the command match statement and add:

Commands::MigrateDb { db_path, backup, dry_run } => {
    use crate::graph::migrate::{migrate_database, check_schema_version};

    if dry_run {
        match check_schema_version(&db_path) {
            Ok(version) => {
                println!("Current schema: v{}", version);
                println!("Target schema: v6");
                if version < 6 {
                    println!("Migration required: yes");
                } else {
                    println!("Migration required: no (already on v6 or later)");
                }
                std::process::exit(0);
            }
            Err(e) => {
                eprintln!("Error checking schema version: {}", e);
                std::process::exit(1);
            }
        }
    }

    match migrate_database(&db_path, backup, false) {
        Ok(result) => {
            if let Some(backup_path) = result.backup_path {
                println!("Backup created: {}", backup_path.display());
            }
            println!("Database migrated: v{} -> v{}", result.previous_version, result.new_version);
            println!("You can now use Magellan 2.0.0 features");
        }
        Err(e) => {
            eprintln!("Migration failed: {}", e);
            std::process::exit(1);
        }
    }
}

Place this after the Export command handler.</action>
