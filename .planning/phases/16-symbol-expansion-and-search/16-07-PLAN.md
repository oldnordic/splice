```rust
fn execute_search(
    pattern: &str,
    path: &Path,
    language: Option<cli::Language>,
    json_output: bool,
) -> Result<splice::cli::CliSuccessPayload> {
    use splice::patch::pattern;

    // Convert CLI language to symbol language
    let symbol_lang = language.map(|l| l.to_symbol_language());

    // Build glob pattern from path
    let glob_pattern = if path.is_dir() {
        format!("{}/**/*.rs", path.display()) // Default to Rust, expand in plan 16-09
    } else {
        path.display().to_string()
    };

    let config = pattern::PatternReplaceConfig {
        glob_pattern,
        find_pattern: pattern.to_string(),
        replace_pattern: String::new(), // Not used for search-only
        language: symbol_lang,
        validate: false,
    };

    let matches = pattern::find_pattern_in_files(&config)?;

    if json_output {
        let results: Vec<Value> = matches.into_iter().map(|m| {
            json!({
                "file": m.file.display(),
                "byte_start": m.byte_start,
                "byte_end": m.byte_end,
                "line": m.line,
                "column": m.column,
                "matched_text": m.matched_text,
            })
        }).collect();

        let payload = serde_json::to_string_pretty(&results)?;
        println!("{}", payload);

        Ok(splice::cli::CliSuccessPayload::already_emitted())
    } else {
        // Human-readable output
        for m in &matches {
            println!("{}:{}:{}: {}", m.file.display(), m.line, m.column, m.matched_text);
        }

        Ok(splice::cli::CliSuccessPayload::message_only(format!(
            "Found {} occurrence(s) of '{}'",
            matches.len(),
            pattern
        )))
    }
}
```
