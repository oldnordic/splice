```rust
pub fn expand_symbol(
    path: &Path,
    byte_offset: usize,
    language: Language,
    level: ExpansionLevel,
) -> Result<(usize, usize)> {
    // Read file and parse with tree-sitter
    let source = std::fs::read(path)?;
    let mut parser = parser_for_language(language)?;
    let tree = parser.parse(&source, None)
        .ok_or_else(|| SpliceError::Other("Parse failed".to_string()))?;

    // Find node at byte offset
    let node = tree.root_node()
        .descendant_for_byte_range(byte_offset, byte_offset)
        .ok_or_else(|| SpliceError::Other("No node at offset".to_string()))?;

    match level {
        ExpansionLevel::None => Ok((byte_offset, byte_offset)),
        ExpansionLevel::Body => {
            let expander: Box<dyn SymbolExpander> = get_expander(language);
            expander.expand_to_body(node, &source)
                .ok_or_else(|| SpliceError::Other("Failed to expand to body".to_string()))
        }
        ExpansionLevel::ContainingBlock => {
            let (start, end) = expand_to_body(node, &source)
                .ok_or_else(|| SpliceError::Other("Failed to expand to body".to_string()))?;
            // Find containing block
            tree_walker::find_containing_block(&tree.root_node(), start, end, &source)
                .ok_or_else(|| SpliceError::Other("Failed to find containing block".to_string()))
        }
    }
}
```

```rust
/// Find the containing block for a given span.
///
/// For methods, finds the containing class/struct.
/// For functions, finds the containing module/impl block.
pub fn find_containing_block(
    root: &Node,
    start: usize,
    end: usize,
    source: &[u8],
) -> Option<(usize, usize)> {
    let mut node = root.descendant_for_byte_range(start, end)?;

    // Walk up the parent chain to find a containing block
    while let Some(parent) = node.parent() {
        let kind = parent.kind();
        match kind {
            // Language-specific containing block kinds
            "class_declaration" | "impl_item" | "trait_item" | "class_definition"
            | "class_specifier" | "struct_specifier" | "namespace_definition"
            | "module_declaration" | "interface_declaration" | "type_alias_declaration" => {
                return Some((parent.start_byte(), parent.end_byte()));
            }
            _ => {
                node = parent;
            }
        }
    }

    None
}
```
