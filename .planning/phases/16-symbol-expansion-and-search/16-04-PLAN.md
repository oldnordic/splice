```rust
/// Extract leading doc comments for a symbol node.
///
/// Walks prev_sibling nodes to find:
/// - Rust: /// (line_comment), //! (line_comment), /**/ (block_comment)
/// - Python: """ (string), # (comment)
/// - C/C++: ///, /**/, //!
/// - Java: /**, ///
/// - JS/TS: /**, ///, ///
///
/// Returns the adjusted start byte offset including docs, or original start if no docs.
pub fn extract_leading_docs(
    node: &Node,
    source: &[u8],
) -> usize {
    let mut current = *node;
    let mut doc_start = node.start_byte();
    let mut found_docs = false;
    let mut blank_lines = 0;

    // Walk previous siblings, stopping at first non-doc, non-blank line
    while let Some(prev) = current.prev_sibling() {
        let kind = prev.kind();
        let is_comment = kind.contains("comment") || kind == "string";

        if is_comment {
            // Check if this looks like a doc comment (starts with ///, /**, //! , /*!, """, ///)
            let text = prev.utf8_text(source).unwrap_or("");
            let is_doc = text.starts_with("///")
                || text.starts_with("/**")
                || text.starts_with("//!")
                || text.starts_with("/*!")
                || text.starts_with("\"\"\"")
                || text.starts_with("///");

            if is_doc {
                doc_start = prev.start_byte();
                found_docs = true;
                blank_lines = 0;
                current = prev;
            } else {
                break;
            }
        } else if kind == "\n" || kind.is_empty() {
            // Allow one blank line between docs and symbol
            blank_lines += 1;
            if blank_lines > 1 {
                break;
            }
            current = prev;
        } else {
            break;
        }
    }

    if found_docs { doc_start } else { node.start_byte() }
}
```

```rust
/// Expand a symbol to its full body including leading doc comments.
///
/// This is the preferred expansion function for user-facing output
/// since documentation provides essential context.
pub fn expand_to_body_with_docs(
    path: &Path,
    byte_offset: usize,
    language: Language,
) -> Result<(usize, usize)> {
    let source = std::fs::read(path)?;
    let mut parser = parser_for_language(language)?;
    let tree = parser.parse(&source, None)
        .ok_or_else(|| SpliceError::Other("Parse failed".to_string()))?;

    let node = tree.root_node()
        .descendant_for_byte_range(byte_offset, byte_offset)
        .ok_or_else(|| SpliceError::Other("No node at offset".to_string()))?;

    // First expand to body
    let expander: Box<dyn SymbolExpander> = get_expander(language);
    let (body_start, body_end) = expander.expand_to_body(node, &source)
        .ok_or_else(|| SpliceError::Other("Failed to expand to body".to_string()))?;

    // Find the body node for doc extraction
    let body_node = tree.root_node()
        .descendant_for_byte_range(body_start, body_end)
        .ok_or_else(|| SpliceError::Other("No body node".to_string()))?;

    // Extend to include leading docs
    let doc_start = tree_walker::extract_leading_docs(&body_node, &source);

    Ok((doc_start, body_end))
}
```
