Update apply_pattern_replace to use atomic writes with rollback:
1. Create backup of each file before modification
2. Use tempfile::NamedTempFile for atomic writes
3. Track all modified files
4. On any error, restore all files from backups
5. On success, clean up backups

Key implementation:
```rust
// Create backup manifest
let mut backups: Vec<(PathBuf, String)> = Vec::new();

for (file_path, file_matches) in matches_by_file {
    // Read and backup original
    let original = std::fs::read_to_string(&file_path)?;
    backups.push((file_path.clone(), original.clone()));

    // Apply replacements
    let mut content = original.clone();
    for m in file_matches {
        content.replace_range(m.byte_start..m.byte_end, &config.replace_pattern);
    }

    // Atomic write using tempfile
    let mut temp = tempfile::NamedTempFile::new_in(file_path.parent().unwrap_or(Path::new(".")))?;
    temp.write_all(content.as_bytes())?;
    temp.persist(&file_path)?;
}

// If we get here, all writes succeeded - return result
```
