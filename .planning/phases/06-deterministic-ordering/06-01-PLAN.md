# Sorting strategy:
- spans: sort by file_path (asc) then byte_start (asc)
- files_affected: sort by file path (asc)
- labels: sort alphabetically (asc)
- symbols (query): sort by file_path then byte_start
- files (apply_files): sort by file path (asc)
- diagnostics: sort by file (asc), then line (asc), then column (asc)

    impl PartialOrd for FilePatternResult {
        fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
            Some(self.cmp(other))
        }
    }
    impl Ord for FilePatternResult {
        fn cmp(&self, other: &Self) -> std::cmp::Ordering {
            self.file.cmp(&other.file)
        }
    }
